import redis.asyncio as aioredis
from celery import Celery
from functools import wraps
import json, logging

logger = logging.getLogger("worker.cache")
celery_app = Celery("classtrack_worker", broker="redis://localhost:6379/0")
redis_client = aioredis.from_url("redis://localhost:6379/1", decode_responses=True)

@celery_app.task(bind=True, max_retries=3, default_retry_delay=10)
def invalidate_user_cache(self, user_id: int):
    """Background worker invalidating user key with exponential backoff"""
    try:
        pattern = f"user:{user_id}:*"
        redis_client.delete(pattern)
        logger.info(f"Cache purge confirmed for user_id={user_id}")
    except Exception as exc:
        logger.warning(f"Cache purge failed; retrying. Error: {exc}")
        raise self.retry(exc=exc)

def cache_response(ttl: int = 3600):
    """Decorator: Cache-Aside with fallback to upstream handler"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            key = f"{func.__name__}:{json.dumps(kwargs, sort_keys=True)}"
            cached = await redis_client.get(key)
            if cached:
                return json.loads(cached)
            result = await func(*args, **kwargs)
            await redis_client.setex(key, ttl, json.dumps(result))
            return result
        return wrapper
    return decorator
